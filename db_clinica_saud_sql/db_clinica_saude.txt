USE db_clinica_saude;

-- 1. Tabela de Pacientes
CREATE TABLE pacientes (
    paciente_id INT AUTO_INCREMENT PRIMARY KEY, -- auto_inc para numerar sequencialmente
    nome VARCHAR(100) NOT NULL, -- até 100 caracteres, obrig preencher
    data_nascimento DATE,
    nif CHAR(9) UNIQUE, -- NIF tem sempre 9 dígitos, nao pode ser repetido
    telemovel VARCHAR(15)
    );

-- 2. Criar Tabela de Médicos
CREATE TABLE medicos (
    medico_id INT AUTO_INCREMENT PRIMARY KEY, -- auto_inc para numerar sequencialmente
    nome VARCHAR(100) NOT NULL, -- até 100 caracteres, obrig preencher
    especialidade VARCHAR(50),
    cedula_profissional VARCHAR(20) UNIQUE -- CD até 15 dígitos, nao pode ser repetido
);

-- 3. Criar Tabela de Consultas 
CREATE TABLE consultas (
    consulta_id INT AUTO_INCREMENT PRIMARY KEY, -- id cons, auto-inc para num sequencial
    paciente_id INT,-- id paciente, igual ao id tabela pacientes
    medico_id INT, -- id medico, igual id tabela medicos
    data_consulta DATETIME,
    diagnostico TEXT,
    valor_consulta DECIMAL(10, 2), -- dez digitos, dos quais dois destinados a casas decimais
    -- linhas abaixo criam o Relacionamentos
    FOREIGN KEY (paciente_id) REFERENCES pacientes(paciente_id),
    FOREIGN KEY (medico_id) REFERENCES medicos(medico_id)
);
